# optional real-model warm-up probe test (T1): configure with an ONNX
# Runtime shared library + a model file to enable it, e.g.
#   -DQSEMSEG_TEST_ORT_LIBRARY=C:/.../onnxruntime.dll
#   -DQSEMSEG_TEST_MODEL=C:/.../model.onnx
# When unset (default, e.g. in CI) the probe test is skipped.
set( QSEMSEG_TEST_ORT_LIBRARY "" CACHE FILEPATH "ONNX Runtime shared library for the TestSemSeg warm-up probe test (optional)" )
set( QSEMSEG_TEST_MODEL "" CACHE FILEPATH "ONNX model file for the TestSemSeg warm-up probe test (optional)" )

# tiny committed CI model (T18): test/data/tiny_fkaconv.onnx (~0.7 MB, same
# graph geometry as the real parity model) + its golden logits
# (test/data/tiny_logits.f32, generated by run_reference.py --prefix tiny).
# When an ONNX Runtime library is configured as well, a TestSemSegRunParity
# test runs the full run-parity against the tiny model - this is how CI
# proves ORT loading + run parity on Linux with zero user hardware. Clear
# the variable to disable the test.
set( QSEMSEG_TINY_MODEL "${CMAKE_CURRENT_LIST_DIR}/data/tiny_fkaconv.onnx" CACHE FILEPATH "Tiny FKAConv ONNX model for the TestSemSegRunParity run-parity test (default: the committed test-data model)" )

add_test( NAME TestSemSeg COMMAND TestSemSeg ${CMAKE_CURRENT_LIST_DIR}/data ${QSEMSEG_TEST_ORT_LIBRARY} ${QSEMSEG_TEST_MODEL} )
set_tests_properties( TestSemSeg PROPERTIES TIMEOUT 600 )

if( QSEMSEG_TINY_MODEL AND QSEMSEG_TEST_ORT_LIBRARY )
	add_test( NAME TestSemSegRunParity
		COMMAND TestSemSeg run
			${CMAKE_CURRENT_LIST_DIR}/data
			${QSEMSEG_TINY_MODEL}
			${QSEMSEG_TEST_ORT_LIBRARY}
			tiny
	)
	set_tests_properties( TestSemSegRunParity PROPERTIES TIMEOUT 600 )
endif()

# SphereFormer run parity (P4): the real 32k export + its golden logits are
# local (multi-100 MB, never committed) - configure with
#   -DQSEMSEG_TEST_SPHERE_MODEL=C:/.../sphereformer_parislille3d_32k.onnx
# (the ORT library comes from QSEMSEG_TEST_ORT_LIBRARY above). Unset (default,
# e.g. in CI) the test is skipped; the no-args sphere stages still run.
set( QSEMSEG_TEST_SPHERE_MODEL "" CACHE FILEPATH "SphereFormer ONNX model for the TestSemSegSphereParity run-parity test (optional, model file is not committed)" )

if( QSEMSEG_TEST_SPHERE_MODEL AND QSEMSEG_TEST_ORT_LIBRARY )
	add_test( NAME TestSemSegSphereParity
		COMMAND TestSemSeg sphere
			${CMAKE_CURRENT_LIST_DIR}/data
			${QSEMSEG_TEST_SPHERE_MODEL}
			${QSEMSEG_TEST_ORT_LIBRARY}
	)
	set_tests_properties( TestSemSegSphereParity PROPERTIES TIMEOUT 600 )
endif()

# self-contained on Windows: the test exe is not run from the install tree, so
# the Qt / MinGW runtime DLLs must sit next to it (otherwise ctest waits on a
# modal 'missing DLL' dialog). The legend regression stage uses ccScalarField
# / ccColorScale, which pulls the whole shared qCC_db -> CCCoreLib ->
# CC_FBO_LIB closure (and its Qt6 Widgets/OpenGL/Concurrent + OpenMP runtime
# dependencies) into the import table at link time.
if( WIN32 )
	add_custom_command( TARGET TestSemSeg POST_BUILD
		COMMAND ${CMAKE_COMMAND} -E copy_if_different
			$<TARGET_FILE:Qt6::Core>
			$<TARGET_FILE:Qt6::Gui>
			$<TARGET_FILE:Qt6::OpenGL>
			$<TARGET_FILE:Qt6::Widgets>
			$<TARGET_FILE:Qt6::Concurrent>
			$<TARGET_FILE:QCC_DB_LIB>
			$<TARGET_FILE:CCCoreLib>
			$<TARGET_FILE:CC_FBO_LIB>
			$<TARGET_FILE_DIR:TestSemSeg>
		COMMAND_EXPAND_LISTS
	)
	if( MINGW )
		get_filename_component( MINGW_BIN_DIR ${CMAKE_CXX_COMPILER} DIRECTORY )
		foreach( MINGW_RUNTIME_DLL libgcc_s_seh-1.dll libstdc++-6.dll libwinpthread-1.dll libgomp-1.dll )
			if( EXISTS ${MINGW_BIN_DIR}/${MINGW_RUNTIME_DLL} )
				add_custom_command( TARGET TestSemSeg POST_BUILD
					COMMAND ${CMAKE_COMMAND} -E copy_if_different ${MINGW_BIN_DIR}/${MINGW_RUNTIME_DLL} $<TARGET_FILE_DIR:TestSemSeg>
					COMMAND_EXPAND_LISTS
				)
			endif()
		endforeach()
	endif()
endif()
